vargist_dismantle_merchant_republic = {
	name = CB_NAME_VARGIST_DISMANTLE_MERCHANT_REPUBLIC
	war_name = WAR_NAME_VARGIST_DISMANTLE_MERCHANT_REPUBLIC
	sprite = 19
	truce_days = 3650
	sort_priority = 1700

	is_permanent = yes
	can_ask_to_join_war = no
	allowed_to_target_tributaries = no

	can_use = {
		ROOT = {
			religion = vargist
			independent = yes
			is_landed = yes
			mercenary = no
			is_merchant_republic = no
			piety = 100
			NOT = { same_realm = FROM }
		}
		FROM = {
			independent = yes
			is_merchant_republic = yes
		}
	}

	is_valid = {
		ROOT = {
			religion = vargist
			independent = yes
		}
		FROM = {
			independent = yes
			is_merchant_republic = yes
		}
	}

	on_add = { ROOT = { piety = -100 } }

	on_success = {
		attacker = {
			prestige = 300
			piety = 250
			participation_scaled_prestige = 200
		}
		defender = {
			prestige = -500
			set_government_type = republic_government
		}
	}

	on_fail = {
		attacker = { prestige = -200 }
		defender = { prestige = 100 }
	}

	on_reverse_demand = {
		attacker = {
			prestige = -400
			transfer_scaled_wealth = { to = defender value = 4.0 }
		}
		defender = { prestige = 200 }
	}

	attacker_ai_victory_worth = { factor = 100 }
	attacker_ai_defeat_worth = { factor = 75 }
	defender_ai_victory_worth = { factor = -1 }
	defender_ai_defeat_worth = { factor = 100 }

	ai_will_do = {
		factor = 0.35
		modifier = { factor = 2 trait = zealous }
		modifier = { factor = 1.5 trait = diligent }
		modifier = { factor = 0.5 trait = content }
	}
}
